no shut
exit

interface fa1/0
ip add 192.168.3.1 255.255.255.252
no shut
exit

interface loopback 0
ip address 1.1.1.1 255.255.255.255
exit

network 192.168.3.0 0.0.0.3 area 0
network 1.1.1.1 0.0.0.0 area 0
exit
exit
debug ip ospf adj

exit
