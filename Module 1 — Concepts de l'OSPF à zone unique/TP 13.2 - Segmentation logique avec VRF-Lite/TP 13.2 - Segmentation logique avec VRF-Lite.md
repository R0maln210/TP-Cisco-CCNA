<img width="882" height="424" alt="01 - lab complet" src="https://github.com/user-attachments/assets/84bb2a41-c3ab-4f89-9250-cc16dc6fc517" />

## R1 : 

```
en
conf t
hostname R1

ip vrf CLIENT-A
rd 65001:1
exit

ip vrf CLIENT-B
rd 65001:2
exit

interface fa1/0
ip vrf forwarding CLIENT-A
ip add 192.168.1.1 255.255.255.0
no shutdown
exit

interface fa2/0
ip vrf forwarding CLIENT-B
ip add 192.168.1.1 255.255.255.0
no shutdown
exit

interface Loopback 0 
ip add 8.8.8.8 255.255.255.255
no shutdown
exit

do wr
```

## PC-CLIENT-A : 

```
set pcname PC-A
ip 192.168.1.10 255.255.255.0 192.168.1.1
save
```

## PC-CLIENT-B : 

```
set pcname PC-B
ip 192.168.1.10 255.255.255.0 192.168.1.1
save
```
